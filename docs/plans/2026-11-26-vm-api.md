# VM API and tiles Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use usask-foundation:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Serve the six Rain Gauge API endpoints and the three rain overlays from the VM's own rasters, and point the web app at them, so the app no longer depends on Google Cloud.

**Architecture:** A small FastAPI service in `server/` reads `/data/rain` (the VM's `<rain dir>/`, read-only) with rasterio and answers in the same formats as the current Cloud Run API. nginx proxies `<base>api/` to it and `<base>tiles/<layer>/{z}/{x}/{y}.png` to the existing TiTiler with a colormap the API computes from `rain_stats.json`. The frontend takes tile templates and legend from `legend_info` and keeps no Google URLs.

**Tech Stack:** Python 3.12, FastAPI, uvicorn, rasterio, numpy, Pillow, pytest, httpx (tests); Docker + docker compose on the server; Vite/React frontend as today.

Spec: `docs/specs/2026-11-26-vm-api-design.md`.

## Global Constraints

- Nothing server-specific (host, paths, ports, data locations) in the repository; all of it comes from `deploy/.env` (ignored by git). Placeholders in templates are `@@NAME@@`.
- Response formats of the current API are kept exactly (field names, string formats, number rounding: 1 decimal for sums, 2 for daily values).
- Crop-year comparison sums count April to November only, for every year alike.
- Breaks: weekly 0 .. p90, season and crop year p5 .. p95, widened to a nice step, at most 8 classes.
- Palette (16 colours, in order): `#e0d8d3 #d8c7b4 #c9ba96 #b0b188 #95a87e #799f76 #5b9770 #388d6d #17816e #05746e #07656b #155766 #20485d #253850 #252a44 #221b38`.
- Python code: PEP 8, type hints, no global mutable state except the stats cache in `RainData`.
- Commit after each task. The repo lives on a Windows drive: after adding a new shell script run `git update-index --chmod=+x <file>`.
- Local Python environment for tests: `~/venv/raingauge-server` (created in Task 1). Run pytest from `server/`.

```python
"""Date windows of the Virtual Rain Gauge and sums over daily series.

Season = Apr 1 .. Oct 31. Crop year cy = Sep 1 (cy-1) .. Aug 31 cy, labelled by the year it ends in.
History before 2026-10 has April to November only, so crop-year comparisons use APRIL_TO_NOVEMBER.
"""
from __future__ import annotations

from datetime import date
from typing import Callable

APRIL_TO_NOVEMBER: frozenset[int] = frozenset(range(4, 12))


def season_window(year: int) -> tuple[date, date]:
    return date(year, 4, 1), date(year, 10, 31)


def crop_year_window(cy: int) -> tuple[date, date]:
    return date(cy - 1, 9, 1), date(cy, 8, 31)


def same_position(newest: date, start: date, end: date) -> date:
    """The day of [start, end] with newest's month and day (Feb 29 -> Feb 28), capped at end."""
    for year in (start.year, end.year):
        day = min(newest.day, 28) if (newest.month == 2 and newest.day == 29) else newest.day
        candidate = date(year, newest.month, day)
        if start <= candidate <= end:
            return candidate
    return end if newest > end else start


def sum_days(
    series: dict[date, float],
    start: date,
    end: date,
    months: frozenset[int] | None = None,
) -> float:
    return float(
        sum(v for d, v in series.items() if start <= d <= end and (months is None or d.month in months))
    )


def comparison_rows(
    series: dict[date, float],
    years: list[int],
    window: Callable[[int], tuple[date, date]],
    newest: date,
    months: frozenset[int] | None,
) -> list[dict]:
    rows = []
    for y in years:
        start, end = window(y)
        to_now = same_position(newest, start, end)
        rows.append(
            {
                "year": y,
                "precip_sum": round(sum_days(series, start, end, months), 1),
                "precip_sum_to_now": round(sum_days(series, start, to_now, months), 1),
            }
        )
    return rows
```

```python
"""Legend entries, class breaks and the TiTiler colormap for the three rain overlays.

Rules as in the original API: weekly 0 .. p90, season and crop year p5 .. p95, widened to a nice
step so there are at most 8 classes. Colours: the original 16-stop ramp (cmocean "rain").
"""
from __future__ import annotations

import io
import json
import math
from urllib.parse import quote

from PIL import Image

PALETTE: list[str] = [
    "#e0d8d3", "#d8c7b4", "#c9ba96", "#b0b188", "#95a87e", "#799f76", "#5b9770", "#388d6d",
    "#17816e", "#05746e", "#07656b", "#155766", "#20485d", "#253850", "#252a44", "#221b38",
]
LAYERS: dict[str, str] = {"weeklyRain": "week", "seasonRain": "season", "yearRain": "year"}
_STEPS = (1, 2, 5, 10, 20, 25, 50, 100, 200, 250, 500, 1000)
_FAR = 1e9


def nice_breaks(lo: float, hi: float, max_classes: int = 8) -> list[float]:
    for step in _STEPS:
        lo_r = math.floor(lo / step) * step
        hi_r = math.ceil(hi / step) * step
        if hi_r <= lo_r:
            hi_r = lo_r + step
        n = round((hi_r - lo_r) / step)
        if n <= max_classes:
            return [float(lo_r + i * step) for i in range(n + 1)]
    return [float(lo), float(hi)]


def _rgba(hex_colour: str) -> list[int]:
    h = hex_colour.lstrip("#")
    return [int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16), 255]


def colormap_query(breaks: list[float]) -> str:
    """TiTiler interval colormap: 16 equal intervals from min to max, open at both ends."""
    lo, hi = breaks[0], breaks[-1]
    n = len(PALETTE)
    edges = [lo + (hi - lo) * i / n for i in range(n + 1)]
    edges[0], edges[-1] = -_FAR, _FAR
    intervals = [[[edges[i], edges[i + 1]], _rgba(PALETTE[i])] for i in range(n)]
    return "colormap=" + quote(json.dumps(intervals, separators=(",", ":")))


def _title(layer: str, stats: dict) -> str:
    if layer == "weeklyRain":
        return "Last weeks precipitation"
    if layer == "seasonRain":
        return "Growing Season Precipitation (April 1 to now)."
    start_year = stats["layers"]["year"]["start"][:4]
    return f"Crop Year Precipitation (Sept {start_year} to now)."


def _breaks_for(layer: str, stats: dict) -> list[float]:
    w = stats["layers"][LAYERS[layer]]
    if layer == "weeklyRain":
        return nice_breaks(0.0, float(w["p90"]))
    return nice_breaks(float(w["p5"]), float(w["p95"]))


def _mm(v: float) -> str:
    return f"{int(v)}mm" if float(v).is_integer() else f"{v:g}mm"


def legend_info(stats: dict, public_base: str) -> dict:
    base = public_base if public_base.endswith("/") else public_base + "/"
    out = {}
    for layer, short in LAYERS.items():
        breaks = _breaks_for(layer, stats)
        out[layer] = {
            "title": _title(layer, stats),
            "uri": f"{base}api/legend/{layer}.png",
            "min": _mm(breaks[0]) if breaks[0] == 0 else "≤" + _mm(breaks[0]),
            "max": "≥" + _mm(breaks[-1]),
            "mode": "continuous",
            "breaks": breaks,
            "palette": PALETTE,
            "tiles": f"{base}tiles/{short}/{{z}}/{{x}}/{{y}}.png?{colormap_query(breaks)}",
            "as_of": stats["newest_day"],
        }
    return out


def legend_png(layer: str) -> bytes:
    """A 256 x 16 bar of the 16 palette colours, left = low."""
    if layer not in LAYERS:
        raise KeyError(layer)
    img = Image.new("RGBA", (256, 16))
    px = img.load()
    for x in range(256):
        r, g, b, a = _rgba(PALETTE[min(x * 16 // 256, 15)])
        for y in range(16):
            px[x, y] = (r, g, b, a)
    buf = io.BytesIO()
    img.save(buf, format="PNG")
    return buf.getvalue()
```

"""A tiny synthetic rain directory with known values.

Grid: 10 x 10 pixels of 0.5 degrees, lon -110 .. -105, lat 55 .. 50 (EPSG:4326).
TARGET pixel (row 2, col 3): every present day has the value of its crop-year stack
(2025 -> 3 mm, 2026 -> 2 mm, 2027 -> 1 mm). NODATA_PIXEL (row 0, col 0) is nodata in every band.
Stacks 2025 and 2026 have April..November days only (like the imported history); stack 2027 has
Sep 1 .. Nov 25 2026 (the newest day), the rest of its bands nodata.
Window COGs at TARGET: week 7, season 214, year 86.
"""
from __future__ import annotations

import json
from datetime import date, timedelta
from pathlib import Path

import numpy as np
import pytest
import rasterio
from rasterio.transform import from_origin

TARGET = (-108.25, 53.75)
OUTSIDE = (-100.0, 53.75)
NODATA_PIXEL = (-109.75, 54.75)
NEWEST = date(2026, 11, 25)
NODATA = -9999.0
_PROFILE = dict(
    driver="GTiff", dtype="float32", width=10, height=10, crs="EPSG:4326",
    transform=from_origin(-110.0, 55.0, 0.5, 0.5), nodata=NODATA,
)


def _days(start: date, end: date):
    d = start
    while d <= end:
        yield d
        d += timedelta(days=1)


def _write(path: Path, bands: list[tuple[str, float | None]], tags: dict) -> None:
    """bands: (description, value at TARGET or None for a nodata band)."""
    with rasterio.open(path, "w", count=len(bands), **_PROFILE) as dst:
        for i, (desc, value) in enumerate(bands, start=1):
            arr = np.full((10, 10), NODATA if value is None else value, dtype="float32")
            arr[0, 0] = NODATA
            dst.write(arr, i)
            dst.set_band_description(i, desc)
        dst.update_tags(**tags)


def _stack(path: Path, cy: int, value: float, present) -> None:
    start, end = date(cy - 1, 9, 1), date(cy, 8, 31)
    bands = [(d.isoformat(), value if present(d) else None) for d in _days(start, end)]
    _write(path, bands, {"crop_year": str(cy), "first_day": start.isoformat(), "units": "mm"})


@pytest.fixture(scope="session")
def rain_dir(tmp_path_factory) -> Path:

```python
"""Point reads of the rain rasters the VM pipeline writes (see docs/specs/2026-11-26-vm-api-design.md).

rain_week/season/year.tif: one band, the window sum. rain_daily_<cy>.tif: one band per day of the
crop year, band description = ISO date, nodata = day missing. rain_stats.json: newest day, crop
years, window dates and percentiles.
"""
from __future__ import annotations

import json
import math
from datetime import date
from pathlib import Path

import rasterio
from rasterio.windows import Window


class DataUnavailable(Exception):
    """A required file is missing or unreadable."""


class RainData:
    def __init__(self, rain_dir: Path) -> None:
        self.rain_dir = Path(rain_dir)
        self._stats: dict | None = None
        self._stats_mtime: float | None = None

    # rain_stats.json, cached until the file changes
    def stats(self) -> dict:
        p = self.rain_dir / "rain_stats.json"
        try:
            mtime = p.stat().st_mtime
            if self._stats is None or mtime != self._stats_mtime:
                self._stats = json.loads(p.read_text())
                self._stats_mtime = mtime
        except (OSError, ValueError) as e:
            raise DataUnavailable(f"rain_stats.json: {e}") from e
        return self._stats

    def newest_day(self) -> date:
        return date.fromisoformat(self.stats()["newest_day"])

    def _column(self, path: Path, lon: float, lat: float) -> tuple[list[float], list[str], float | None] | None:
        """All band values at (lon, lat), the band descriptions and nodata; None if outside the grid."""
        try:
            with rasterio.open(path) as ds:
                row, col = ds.index(lon, lat)
                if not (0 <= row < ds.height and 0 <= col < ds.width):
                    return None
                values = ds.read(window=Window(col, row, 1, 1))[:, 0, 0]
                return [float(v) for v in values], list(ds.descriptions), ds.nodata
        except rasterio.errors.RasterioIOError as e:
            raise DataUnavailable(f"{path.name}: {e}") from e

    @staticmethod
    def _present(v: float, nodata: float | None) -> bool:
        return not math.isnan(v) and (nodata is None or v != nodata)

    def window_value(self, short: str, lon: float, lat: float) -> float | None:
        col = self._column(self.rain_dir / f"rain_{short}.tif", lon, lat)
        if col is None:
            return None
        values, _, nodata = col
        return values[0] if self._present(values[0], nodata) else None

    def stack_series(self, cy: int, lon: float, lat: float) -> dict[date, float]:
        path = self.rain_dir / f"rain_daily_{cy}.tif"
        if not path.exists():
            return {}
        col = self._column(path, lon, lat)
        if col is None:
            return {}
        values, descriptions, nodata = col
        out: dict[date, float] = {}
        for v, desc in zip(values, descriptions):
            if desc and self._present(v, nodata):
                out[date.fromisoformat(desc[:10])] = v
        return out

    def series_for_years(self, crop_years: list[int], lon: float, lat: float) -> dict[date, float]:
        merged: dict[date, float] = {}
        for cy in crop_years:
            merged.update(self.stack_series(cy, lon, lat))
        return merged
```

```python
"""Virtual Rain Gauge API served from the VM's own rasters.

Same request and response formats as the original Cloud Run API (api.realfield.ca), so the web app
and later the mobile app can use either.
"""
from __future__ import annotations

from datetime import date, timedelta

from fastapi import FastAPI, HTTPException, Request, Response
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse
from pydantic import BaseModel, Field

from .config import Settings, settings_from_env
from .legend import LAYERS, legend_info, legend_png
from .rain_data import DataUnavailable, RainData
from .windows import APRIL_TO_NOVEMBER, comparison_rows, crop_year_window, season_window

YEARS_SHOWN = 7
NO_DATA = HTTPException(status_code=404, detail="No data at this location")


class Point(BaseModel):
    lat: float = Field(ge=-90, le=90)
    lon: float = Field(ge=-180, le=180)


def _month(d: date) -> str:
    return "Sept" if d.month == 9 else d.strftime("%b")


def _window_text(stats: dict, short: str, value: float) -> str:
    w = stats["layers"][short]
    start = date.fromisoformat(w["start"])
    end = date.fromisoformat(w["end"])
    end_text = "today" if w["end"] == stats["newest_day"] else f"{_month(end)} {end.day:02d}"
    return f"{_month(start)} {start.day:02d} - {end_text}: {value:.1f}mm"


def create_app(settings: Settings) -> FastAPI:
    app = FastAPI(title="Virtual Rain Gauge API", docs_url=None, redoc_url=None)
    app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["GET", "POST"], allow_headers=["*"])
    data = RainData(settings.rain_dir)

    @app.exception_handler(DataUnavailable)
    async def _unavailable(request: Request, exc: DataUnavailable) -> JSONResponse:
        return JSONResponse(status_code=503, content={"detail": "Rain data unavailable"})

    @app.get("/health")
    def health() -> dict:
        return {"ok": True, "newest_day": data.stats()["newest_day"]}

    @app.get("/version")
    def version() -> dict:
        return {
            "web": {"minimumRequiredVersion": settings.app_version},
            "ios": {"minimumRequiredVersion": "1.0.6", "minimumBuildNumber": "1",
                    "storeUrl": "https://apps.apple.com/app/virtual-rain-gauge"},
            "android": {"minimumRequiredVersion": "1.0.6", "minimumVersionCode": 1,
                        "storeUrl": "https://play.google.com/store/apps/details?id=com.namtranhoai.rain_gauge_app"},
        }

    @app.get("/legend_info")
    def legend() -> dict:
        return legend_info(data.stats(), settings.public_base)

    @app.get("/legend/{layer}.png")
    def legend_image(layer: str) -> Response:
        if layer not in LAYERS:
            raise HTTPException(status_code=404, detail="Unknown layer")
        return Response(content=legend_png(layer), media_type="image/png",
                        headers={"Cache-Control": "public, max-age=86400"})

    @app.post("/precipitation_values")
    def precipitation_values(p: Point) -> dict:
        stats = data.stats()
        values = {short: data.window_value(short, p.lon, p.lat) for short in ("week", "season", "year")}
        if all(v is None for v in values.values()):
            raise NO_DATA
        week = values["week"] or 0.0
        return {
            "weekly_precipitation": f"Last 7 days: {week:.1f}mm",
            "seasonal_precipitation": _window_text(stats, "season", values["season"] or 0.0),
            "yearly_precipitation": _window_text(stats, "year", values["year"] or 0.0),
        }

    @app.post("/daily_precipitation")
    def daily_precipitation(p: Point) -> dict:
        stats = data.stats()
        newest = data.newest_day()
        series = data.stack_series(stats["crop_years"][-1], p.lon, p.lat)
        if not series:
            raise NO_DATA
        out = {}
        for i in range(7):
            d = newest - timedelta(days=i)
            if d in series:
                out[d.isoformat()] = {"precip": round(series[d], 2)}
        return out

    def _comparison(p: Point, kind: str) -> dict:
        stats = data.stats()
        newest = data.newest_day()
        current_cy = stats["crop_years"][-1]
        if kind == "season":
            current = int(stats["layers"]["season"]["start"][:4])
            years = list(range(current - YEARS_SHOWN + 1, current + 1))
            stacks = list(range(years[0], current_cy + 1))  # season Y spans stacks Y and Y+1
            window, months = season_window, None
        else:
            years = list(range(current_cy - YEARS_SHOWN + 1, current_cy + 1))
            stacks = years
            window, months = crop_year_window, APRIL_TO_NOVEMBER
        series = data.series_for_years(stacks, p.lon, p.lat)
        if not series:
            raise NO_DATA
        return {"data": comparison_rows(series, years, window, newest, months)}

    @app.post("/season_comparison")
    def season_comparison(p: Point) -> dict:
        return _comparison(p, "season")

    @app.post("/year_comparison")
    def year_comparison(p: Point) -> dict:
        return _comparison(p, "year")

    return app


app = create_app(settings_from_env())
```

```javascript
import { useEffect, useState } from 'react';
import API_CONFIG from '../config/apiConfig';

/**
 * Fetches /legend_info once: titles, breaks, palette, legend image and the tile URL template of
 * each overlay. The map and the legend both read it, so they always agree.
 */
export function useLegendInfo() {
  const [info, setInfo] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const response = await fetch(
          `${API_CONFIG.BASE_URL}${API_CONFIG.ENDPOINTS.LEGEND_INFO}`
        );
        const data = await response.json();
        if (!cancelled) setInfo(data);
      } catch (error) {
        console.error('Error fetching legend:', error);
      } finally {
        if (!cancelled) setLoading(false);
      }
    })();
    return () => {
      cancelled = true;
    };
  }, []);

  return { info, loading };
}
```

- [ ] **Step 1: Write `server/tools/compare_api.py`**

```python
"""Compare two Rain Gauge APIs at a list of points.

    python compare_api.py <old api> https://<server>/raingauge/api

Prints, per point: the three window totals from each API, and per comparison chart the per-year
difference. Weekly totals are expected to agree to about 0.1 mm; history differs by source.
"""
from __future__ import annotations

import json
import re
import sys
import urllib.request

POINTS = {
    "Saskatoon": (52.13, -106.67),
    "Regina": (50.45, -104.62),
    "Swift Current": (50.29, -107.80),
    "Yorkton": (51.21, -102.46),
    "Calgary": (51.05, -114.07),
    "Winnipeg": (49.90, -97.14),
    "Lethbridge": (49.70, -112.83),
    "Prince Albert": (53.20, -105.75),
}


def post(base: str, path: str, lat: float, lon: float) -> dict:
    req = urllib.request.Request(
        f"{base}/{path}", data=json.dumps({"lat": lat, "lon": lon}).encode(),
        headers={"Content-Type": "application/json"}, method="POST",
    )
    with urllib.request.urlopen(req, timeout=30) as r:
        return json.load(r)


def mm(text: str) -> float:
    m = re.search(r"([\d.]+)mm", text)
    return float(m.group(1)) if m else float("nan")


def main(old: str, new: str) -> None:
    print(f"{'point':14} {'week old/new':>16} {'season old/new':>18} {'year old/new':>16}")
    for name, (lat, lon) in POINTS.items():
        a = post(old, "precipitation_values", lat, lon)
        b = post(new, "precipitation_values", lat, lon)
        print(
            f"{name:14} {mm(a['weekly_precipitation']):7.1f}/{mm(b['weekly_precipitation']):<7.1f} "
            f"{mm(a['seasonal_precipitation']):8.1f}/{mm(b['seasonal_precipitation']):<8.1f} "
            f"{mm(a['yearly_precipitation']):7.1f}/{mm(b['yearly_precipitation']):<7.1f}"
        )
    for chart in ("season_comparison", "year_comparison"):
        print(f"\n{chart}: precip_sum old -> new (difference)")
        for name, (lat, lon) in POINTS.items():
            a = {int(r["year"]): r["precip_sum"] for r in post(old, chart, lat, lon)["data"]}
            b = {int(r["year"]): r["precip_sum"] for r in post(new, chart, lat, lon)["data"]}
            cells = [f"{y}: {a[y]:.0f}->{b[y]:.0f} ({b[y]-a[y]:+.0f})" for y in sorted(set(a) & set(b))]
            print(f"  {name:14} " + "  ".join(cells))


if __name__ == "__main__":
    if len(sys.argv) != 3:
        sys.exit(__doc__)
    main(sys.argv[1].rstrip("/"), sys.argv[2].rstrip("/"))
```
